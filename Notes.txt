Set Git path in Power Shell
+++++++++++++++++++++++++++++++++++++++++
$gitPath = "I:\Program Files\Git\cmd"
$userPath = [Environment]::GetEnvironmentVariable("Path", "User")

if (($userPath -split ';') -notcontains $gitPath) {
    [Environment]::SetEnvironmentVariable(
        "Path",
        "$userPath;$gitPath",
        "User"
    )
}

$env:Path += ";$gitPath"
git --version
